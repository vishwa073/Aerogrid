import React, { useEffect, useState } from 'react';

import { CutoffAuditTrail } from './components/dashboard/CutoffAuditTrail';
import {
  DashboardError,
  DashboardLoading,
} from './components/dashboard/DashboardStates';
import { ErrorAnalytics } from './components/dashboard/ErrorAnalytics';
import { GenerationChart } from './components/dashboard/GenerationChart';

import { useEvaluation } from './hooks/useEvaluation';
import { api } from './services/api';
import type { HealthResponse } from './types/api';
import { IntervalTable } from './components/dashboard/IntervalsTable';

export const App: React.FC = () => {
  const [health, setHealth] = useState<HealthResponse | null>(null);
  const [healthError, setHealthError] = useState<string | null>(null);
  const [currentView, setCurrentView] = useState('dashboard');

  const { evaluation, loading, error, retry } = useEvaluation(11);

  useEffect(() => {
    let isActive = true;

    api
      .checkHealth()
      .then((data) => {
        if (isActive) {
          setHealth(data);
          setHealthError(null);
        }
      })
      .catch((err: unknown) => {
        if (isActive) {
          setHealthError(
            err instanceof Error
              ? err.message
              : 'Failed to connect to backend',
          );
        }
      });

    return () => {
      isActive = false;
    };
  }, []);

  const renderEvaluationState = (emptyMessage: string) => {
    if (loading) {
      return <DashboardLoading />;
    }

    if (error) {
      return <DashboardError message={error} onRetry={retry} />;
    }

    if (!evaluation) {
      return (
        <section className="rounded-xl border border-slate-200 bg-white p-6 text-sm text-slate-600">
          {emptyMessage}
        </section>
      );
    }

    return null;
  };

  const renderDashboard = () => {
    const evaluationState = renderEvaluationState(
      'No evaluation data is available for this period.',
    );

    if (evaluationState) {
      return (
        <>
          <section>
            <div className="flex flex-wrap items-start justify-between gap-3">
              <div>
                <p className="text-xs font-semibold uppercase tracking-wider text-emerald-700">
                  UK wind generation
                </p>
                <h1 className="mt-1 text-2xl font-bold tracking-tight text-slate-900">
                  Forecast Overview
                </h1>
                <p className="mt-2 max-w-2xl text-sm leading-relaxed text-slate-600">
                  Understand how day-ahead wind forecasts compared with actual
                  UK generation throughout January 2026.
                </p>
              </div>

              <span className="rounded-lg border border-slate-200 bg-white px-3 py-2 text-xs font-medium text-slate-600">
                Cutoff: 11:00 UTC
              </span>
            </div>
          </section>

          {evaluationState}
        </>
      );
    }

    if (!evaluation) {
      return null;
    }

    return (
      <>
        <section>
          <div className="flex flex-wrap items-start justify-between gap-3">
            <div>
              <p className="text-xs font-semibold uppercase tracking-wider text-emerald-700">
                UK wind generation
              </p>
              <h1 className="mt-1 text-2xl font-bold tracking-tight text-slate-900">
                Forecast Overview
              </h1>
              <p className="mt-2 max-w-2xl text-sm leading-relaxed text-slate-600">
                Understand how day-ahead wind forecasts compared with actual UK
                generation throughout January 2026.
              </p>
            </div>

            <span className="rounded-lg border border-slate-200 bg-white px-3 py-2 text-xs font-medium text-slate-600">
              Cutoff: 11:00 UTC
            </span>
          </div>
        </section>

        <section
          aria-label="Evaluation summary"
          className="grid grid-cols-1 gap-3 sm:grid-cols-3"
        >
          <div className="rounded-xl border border-slate-200 bg-white p-4">
            <p className="text-xs font-medium text-slate-500">
              Comparison resolution
            </p>
            <p className="mt-2 text-lg font-semibold text-slate-900">Hourly</p>
            <p className="mt-1 text-xs text-slate-500">
              Actuals aggregated from half-hourly readings
            </p>
          </div>

          <div className="rounded-xl border border-slate-200 bg-white p-4">
            <p className="text-xs font-medium text-slate-500">
              Matched intervals

            <p className="mt-2 text-lg font-semibold text-slate-900">
              {evaluation.matched_interval_count.toLocaleString('en-GB')}
            </p>
            <p className="mt-1 text-xs text-slate-500">
              Matched forecast and actual pairs
            </p>
          </div>

          <div className="rounded-xl border border-slate-200 bg-white p-4">
            <p className="text-xs font-medium text-slate-500">
              Evaluation period
            </p>
            <p className="mt-2 text-lg font-semibold text-slate-900">
              {evaluation.period.start} – {evaluation.period.end}
            </p>
            <p className="mt-1 text-xs text-slate-500">
              UK national wind generation
            </p>
          </div>
        </section>

        <KpiGrid metrics={evaluation.metrics} />
        <GenerationChart intervals={evaluation.intervals} />
        <CutoffAuditTrail evaluation={evaluation} />

            hourly intervals.
          </p>
        </section>
      </>
    );
  };

  const renderSelectedView = () => {
    if (currentView === 'accuracy') {
      const evaluationState = renderEvaluationState(
        'No evaluation data is available for Error Analytics.',

      if (evaluationState) {
        return evaluationState;

        </p>
      </div>

      <IntervalTable intervals={evaluation.intervals} />
    </section>
  ) : null;
}

    return renderDashboard();
  };

  return (
    <AppLayout
      health={health}
      healthError={healthError}
      currentView={currentView}
      onNavigate={setCurrentView}
    >
      {renderSelectedView()}
      <Footer />
    </AppLayout>
  );
};

export default App;

import { Activity, TrendingDown, TrendingUp } from 'lucide-react';
import type { EvaluationResponse } from '../../types/api';

interface ErrorAnalyticsProps {
  evaluation: EvaluationResponse;
}

export function ErrorAnalytics({ evaluation }: ErrorAnalyticsProps) {
  const { intervals, metrics } = evaluation;

  const errors = intervals.map((interval) => ({
    time: interval.start_time,
    error: interval.actual_mw - interval.forecast_mw,
  }));

  const overForecastCount = errors.filter((item) => item.error < 0).length;
  const underForecastCount = errors.filter((item) => item.error > 0).length;
  const largestError = errors.reduce(
    (largest, item) =>
      Math.abs(item.error) > Math.abs(largest.error) ? item : largest,
    { time: '', error: 0 },
  );

  const formatMw = (value: number) =>
    `${new Intl.NumberFormat('en-GB', {
      maximumFractionDigits: 1,
    }).format(value)} MW`;

  return (
    <div className="space-y-6">
      <div>
        <p className="text-sm font-medium text-slate-500">
          January 2026 / Forecast performance
        </p>
        <h1 className="mt-1 text-2xl font-bold text-slate-900">
          Error Analytics
        </h1>
        <p className="mt-2 max-w-3xl text-sm leading-6 text-slate-600">
          Analyse the difference between actual and forecast wind generation.
          Positive error means actual generation exceeded the forecast;
          negative error means the forecast exceeded actual generation.
        </p>
      </div>

      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-3">
        <article className="rounded-xl border border-slate-200 bg-white p-5 shadow-sm">
          <div className="flex items-center gap-2 text-sm text-slate-500">
            <Activity size={18} />
            Mean absolute error
          </div>
          <p className="mt-3 text-2xl font-bold text-slate-900">
            {formatMw(metrics.mae_mw)}
          </p>
          <p className="mt-2 text-xs text-slate-500">
            Average magnitude of forecast error

        </article>

          <div className="flex items-center gap-2 text-sm text-slate-500">
            <TrendingDown size={18} />

            {overForecastCount.toLocaleString('en-GB')}
          </p>
          <p className="mt-2 text-xs text-slate-500">
            Forecast generation was above actual generation

          </p>

          <div>
            <p className="text-sm text-slate-500">Absolute error</p>

              {formatMw(Math.abs(largestError.error))}

                          .slice(0, 16)}
                      </td>

                          maximumFractionDigits: 1,
                        })}
                      </td>
                      <td className="px-5 py-3 text-right tabular-nums">
                        {interval.forecast_mw.toLocaleString('en-GB', {
                          maximumFractionDigits: 1,
                        })}
                      </td>
                      <td

                      >
                        {error > 0 ? '+' : ''}
                        {error.toLocaleString('en-GB', {
                          maximumFractionDigits: 1,
                        })}
                      </td>
                    </tr>
                  );
                })}
            </tbody>
          </table>
        </div>
      </section>
    </div>
  );
}

import {
  CartesianGrid,
  Legend,
  Line,
  LineChart,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from 'recharts';

import type { EvaluationInterval } from '../../types/api';

interface GenerationChartProps {
  intervals: EvaluationInterval[];
}

interface ChartPoint {
  timestamp: string;
  actual: number;
  forecast: number;
}

function formatTimestamp(value: string): string {
  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return date.toLocaleString('en-GB', {
    day: '2-digit',
    month: 'short',
    hour: '2-digit',
    minute: '2-digit',
    timeZone: 'UTC',
    hour12: false,
  });
}

function formatMw(value: number): string {
  return `${new Intl.NumberFormat('en-GB', {
    maximumFractionDigits: 0,
  }).format(value)} MW`;
}

export function GenerationChart({ intervals }: GenerationChartProps) {

import { useEffect, useState } from 'react';
import { api } from '../services/api';
import type { EvaluationResponse } from '../types/api';

interface UseEvaluationResult {
  evaluation: EvaluationResponse | null;
  loading: boolean;
  error: string | null;
  retry: () => void;
}

export function useEvaluation(cutoffHourUtc = 11): UseEvaluationResult {
  const [evaluation, setEvaluation] = useState<EvaluationResponse | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [refreshKey, setRefreshKey] = useState(0);

  useEffect(() => {
    let isActive = true;

    setLoading(true);
    setError(null);

    api.getJanuaryEvaluation(cutoffHourUtc)
      .then((data) => {
        if (isActive) setEvaluation(data);
      })
      .catch((err: unknown) => {
        if (isActive) {
          setError(
            err instanceof Error
              ? err.message
              : 'Failed to load the January evaluation',
          );
        }
      })
      .finally(() => {
        if (isActive) setLoading(false);
      });

    return () => {
      isActive = false;
    };
  }, [cutoffHourUtc, refreshKey]);

  return {
    evaluation,
    loading,
    error,
    retry: () => setRefreshKey((current) => current + 1),
  };
}

import type {
  EvaluationResponse,
  HealthResponse,
} from '../types/api';

const API_BASE_URL = (
  import.meta.env.VITE_API_BASE_URL || ''
).replace(/\/$/, '');

class ApiService {
  private baseUrl: string;

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl;
  }

  async checkHealth(): Promise<HealthResponse> {
    const response = await fetch(`${this.baseUrl}/health`);

    if (!response.ok) {
      throw new Error(
        `Health check failed with status: ${response.status}`
      );
    }

    return response.json() as Promise<HealthResponse>;
  }

  async getJanuaryEvaluation(
    cutoffHourUtc = 11
  ): Promise<EvaluationResponse> {
    const params = new URLSearchParams({
      cutoff_hour_utc: String(cutoffHourUtc),
    });

    const response = await fetch(
      `${this.baseUrl}/api/v1/evaluation/january-2026?${params}`
    );

    if (!response.ok) {
      let message = `Evaluation request failed with status: ${response.status}`;

      try {
        const errorBody = await response.json();
        if (typeof errorBody.detail === 'string') {
          message = errorBody.detail;
        }
      } catch {
        // Keep the original status-based error message.
      }

      throw new Error(message);
    }

    return response.json() as Promise<EvaluationResponse>;
  }
}

export const api = new ApiService(API_BASE_URL);
